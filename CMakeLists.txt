cmake_minimum_required(VERSION 3.25)
project(FreedomControl VERSION 0.5.5 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Use Windows x64 with MSVC (Visual Studio 2022/2026). Portable tests can be built from tests/ separately.")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(COMMONLIBSSE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../CommonLibSSE-NG" CACHE PATH "Local CommonLibSSE-NG checkout")
if(NOT EXISTS "${COMMONLIBSSE_PATH}/include/RE/Skyrim.h")
    message(FATAL_ERROR "COMMONLIBSSE_PATH does not contain CommonLibSSE-NG headers: ${COMMONLIBSSE_PATH}")
endif()
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(REX_OPTION_INI OFF CACHE BOOL "" FORCE)
set(REX_OPTION_JSON OFF CACHE BOOL "" FORCE)
set(REX_OPTION_TOML OFF CACHE BOOL "" FORCE)
set(SKSE_SUPPORT_XBYAK OFF CACHE BOOL "" FORCE)
set(SKSE_SUPPORT_PATCH_SAFETY OFF CACHE BOOL "" FORCE)
set(COMMONLIB_PREBUILT OFF CACHE BOOL "" FORCE)
set(COMMONLIB_PREBUILT_MULTICONFIG OFF CACHE BOOL "" FORCE)
set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "" FORCE)
add_subdirectory("${COMMONLIBSSE_PATH}" "${CMAKE_CURRENT_BINARY_DIR}/_commonlib" EXCLUDE_FROM_ALL)
find_package(imgui CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(fmt CONFIG REQUIRED)
find_package(spdlog CONFIG REQUIRED)
find_package(Git QUIET)
set(FC_COMMONLIB_COMMIT "unknown-local-checkout")
if(GIT_FOUND)
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${COMMONLIBSSE_PATH}" rev-parse HEAD
        OUTPUT_VARIABLE FC_GIT_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE FC_GIT_RESULT)
    if(FC_GIT_RESULT EQUAL 0 AND FC_GIT_COMMIT MATCHES "^[0-9a-fA-F]+$")
        set(FC_COMMONLIB_COMMIT "${FC_GIT_COMMIT}")
    endif()
endif()
add_library(FreedomControl SHARED src/VMArgumentsContract18.cpp src/ApiContract17.cpp src/Plugin.cpp src/Engine.cpp src/PlayerCustomization.cpp src/Input13.cpp src/GameInput13.cpp src/Legion.cpp src/Follower10.cpp src/Spawner10.cpp src/WorldFreedom10.cpp src/Kernel11.cpp src/PlayerFreedom14.cpp src/QuestCenter11.cpp src/KernelSave11.cpp src/CrimeHooks11.cpp src/SexLab16.cpp src/Overlay.cpp src/Core.cpp)
include(cmake/SourceIdentity.cmake)
fc_apply_build_identity(FreedomControl "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(FreedomControl PRIVATE include src)
target_precompile_headers(FreedomControl PRIVATE src/PCH.h)
set_source_files_properties(src/Core.cpp PROPERTIES SKIP_PRECOMPILE_HEADERS ON)
target_compile_options(FreedomControl PRIVATE /utf-8 /permissive- /EHsc /W3 /bigobj)
target_compile_definitions(FreedomControl PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX
    "FC_COMMONLIB_COMMIT=\"${FC_COMMONLIB_COMMIT}\"")
target_link_libraries(FreedomControl PRIVATE CommonLibSSE::CommonLibSSE imgui::imgui
    nlohmann_json::nlohmann_json fmt::fmt spdlog::spdlog
    d3d11 dxgi d3dcompiler user32 imm32 dwmapi)
set_target_properties(FreedomControl PROPERTIES PREFIX "" OUTPUT_NAME "FreedomControl")
install(TARGETS FreedomControl RUNTIME DESTINATION SKSE/Plugins COMPONENT Runtime)
install(FILES package/SKSE/Plugins/FreedomControl.ini DESTINATION SKSE/Plugins COMPONENT Runtime)
install(FILES package/FreedomControlRuntime.esp DESTINATION . COMPONENT Runtime)
install(FILES package/SKSE/Plugins/FreedomControl.creatures.json DESTINATION SKSE/Plugins COMPONENT Runtime)
option(FC_BUILD_PORTABLE_TESTS "Build engine-independent unit tests" ON)
if(FC_BUILD_PORTABLE_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
message(STATUS "FreedomControl: experimental 1.6.1170-only source build; CommonLib commit ${FC_COMMONLIB_COMMIT}")
