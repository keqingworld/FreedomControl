cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(FreedomControlPortableTests LANGUAGES CXX)
    enable_testing()
endif()
add_executable(fc_core_tests core_tests.cpp ../src/Core.cpp)
target_include_directories(fc_core_tests PRIVATE ../include)
target_compile_features(fc_core_tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(fc_core_tests PRIVATE /utf-8 /W4 /EHsc)
else()
    target_compile_options(fc_core_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME fc_core COMMAND fc_core_tests)
